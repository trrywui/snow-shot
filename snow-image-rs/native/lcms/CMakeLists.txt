cmake_minimum_required(VERSION 3.20)
project(snow_lcms LANGUAGES C)
file(GLOB SOURCES CONFIGURE_DEPENDS "${SNOW_SOURCE}/src/*.c")
add_library(lcms2 STATIC ${SOURCES})
target_include_directories(lcms2 PUBLIC "${SNOW_SOURCE}/include")
set_target_properties(lcms2 PROPERTIES C_STANDARD 99)
install(TARGETS lcms2 ARCHIVE DESTINATION lib)
install(FILES "${SNOW_SOURCE}/include/lcms2.h" "${SNOW_SOURCE}/include/lcms2_plugin.h" DESTINATION include)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lcms2.pc" "prefix=${CMAKE_INSTALL_PREFIX}\nlibdir=\${prefix}/lib\nincludedir=\${prefix}/include\nName: lcms2\nDescription: Little CMS\nVersion: 2.17\nLibs: -L\${libdir} -llcms2\nLibs.private: -lm\nCflags: -I\${includedir}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/lcms2.pc" DESTINATION lib/pkgconfig)
