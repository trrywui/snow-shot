cmake_minimum_required(VERSION 3.20)
project(snow_image_native LANGUAGES C CXX)
add_library(snow_image_native STATIC native.cpp jpeg.c)
target_include_directories(snow_image_native PRIVATE "${SNOW_NATIVE_PREFIX}/include")
target_compile_features(snow_image_native PRIVATE cxx_std_17)
set_target_properties(snow_image_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_definitions(snow_image_native PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(snow_image_native PRIVATE -Wall -Wextra -Werror)
endif()

target_compile_definitions(snow_image_native PRIVATE JXL_STATIC_DEFINE JXL_THREADS_STATIC_DEFINE JXL_CMS_STATIC_DEFINE)
set(SNOW_SANITIZERS "" CACHE STRING "Clang sanitizers for native adapter validation")
if(SNOW_SANITIZERS)
  target_compile_options(snow_image_native PRIVATE "-fsanitize=${SNOW_SANITIZERS}" -fno-omit-frame-pointer)
  target_link_options(snow_image_native INTERFACE "-fsanitize=${SNOW_SANITIZERS}")
endif()
option(SNOW_BUILD_TESTS "Build only the native adapter lifecycle harness" OFF)
if(SNOW_BUILD_TESTS)
  enable_testing()
  add_executable(snow_native_lifecycle lifecycle.cpp)
  target_link_libraries(snow_native_lifecycle PRIVATE snow_image_native)
  foreach(LIB jpeg webpmux webpdemux webp sharpyuv jxl jxl_threads jxl_cms hwy brotlienc brotlidec brotlicommon avif aom dav1d lcms2 zimg)
    find_library(SNOW_${LIB} NAMES ${LIB} ${LIB}-static PATHS "${SNOW_NATIVE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(snow_native_lifecycle PRIVATE "${SNOW_${LIB}}")
  endforeach()
  if(SNOW_SANITIZERS)
    target_compile_options(snow_native_lifecycle PRIVATE "-fsanitize=${SNOW_SANITIZERS}" -fno-omit-frame-pointer)
  endif()
  add_test(NAME snow-native-lifecycle COMMAND snow_native_lifecycle)
endif()
option(SNOW_BUILD_FUZZER "Build the native decoder/animation libFuzzer harness" OFF)
if(SNOW_BUILD_FUZZER)
  # Apple Clang ships sanitizer instrumentation but not the libFuzzer runtime.
  # An explicitly provided, locked cargo-fuzz libfuzzer-sys source is sufficient.
  set(SNOW_LIBFUZZER_SOURCE "" CACHE PATH "Optional prepared libFuzzer source directory")
  add_executable(snow_native_fuzz fuzz.cpp)
  target_link_libraries(snow_native_fuzz PRIVATE snow_image_native)
  foreach(LIB jpeg webpmux webpdemux webp sharpyuv jxl jxl_threads jxl_cms hwy brotlienc brotlidec brotlicommon avif aom dav1d lcms2 zimg)
    find_library(SNOW_${LIB} NAMES ${LIB} ${LIB}-static PATHS "${SNOW_NATIVE_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(snow_native_fuzz PRIVATE "${SNOW_${LIB}}")
  endforeach()
  target_compile_options(snow_image_native PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_compile_options(snow_native_fuzz PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  if(SNOW_LIBFUZZER_SOURCE)
    if(NOT EXISTS "${SNOW_LIBFUZZER_SOURCE}/FuzzerMain.cpp")
      message(FATAL_ERROR "SNOW_LIBFUZZER_SOURCE must contain prepared libFuzzer C++ sources")
    endif()
    file(GLOB SNOW_FUZZER_SOURCES CONFIGURE_DEPENDS "${SNOW_LIBFUZZER_SOURCE}/*.cpp")
    add_library(snow_fuzzer_runtime STATIC ${SNOW_FUZZER_SOURCES})
    target_compile_features(snow_fuzzer_runtime PRIVATE cxx_std_17)
    # Keep libc++ container annotations consistent with the instrumented adapter.
    # Do not instrument the fuzzing runtime with coverage (which would recurse).
    target_compile_options(snow_fuzzer_runtime PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_libraries(snow_native_fuzz PRIVATE snow_fuzzer_runtime)
    target_link_options(snow_native_fuzz PRIVATE -fsanitize=address,undefined)
  else()
    target_link_options(snow_native_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  endif()
endif()
