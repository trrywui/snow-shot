cmake_minimum_required(VERSION 3.20)
project(snow_zimg LANGUAGES CXX)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${SNOW_SOURCE}/src/zimg/*.cpp")
list(FILTER SOURCES EXCLUDE REGEX "/(x86|arm)/")
add_library(zimg STATIC ${SOURCES})
target_include_directories(zimg PRIVATE "${SNOW_SOURCE}/src/zimg")
target_compile_features(zimg PRIVATE cxx_std_14)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64|ARM64")
  file(GLOB_RECURSE SIMD CONFIGURE_DEPENDS "${SNOW_SOURCE}/src/zimg/*/arm/*.cpp")
  target_sources(zimg PRIVATE ${SIMD})
  target_compile_definitions(zimg PRIVATE ZIMG_ARM)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
  file(GLOB_RECURSE SIMD CONFIGURE_DEPENDS "${SNOW_SOURCE}/src/zimg/*/x86/*.cpp")
  list(FILTER SIMD EXCLUDE REGEX "avx512")
  target_sources(zimg PRIVATE ${SIMD})
  target_compile_definitions(zimg PRIVATE ZIMG_X86)
  foreach(SOURCE IN LISTS SIMD)
    if(SOURCE MATCHES "avx2")
      set(FLAGS -mavx2 -mf16c -mfma)
    elseif(SOURCE MATCHES "ivb")
      set(FLAGS -mavx -mf16c)
    elseif(SOURCE MATCHES "avx")
      set(FLAGS -mavx)
    elseif(SOURCE MATCHES "sse2")
      set(FLAGS -msse2)
    elseif(SOURCE MATCHES "sse")
      set(FLAGS -msse)
    else()
      set(FLAGS "")
    endif()
    if(MSVC)
      list(TRANSFORM FLAGS PREPEND "/clang:")
    endif()
    set_source_files_properties(${SOURCE} PROPERTIES COMPILE_OPTIONS "${FLAGS}")
  endforeach()
endif()
install(TARGETS zimg ARCHIVE DESTINATION lib)
install(FILES "${SNOW_SOURCE}/src/zimg/api/zimg.h" DESTINATION include)
