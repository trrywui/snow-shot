cmake_minimum_required(VERSION 3.20)
project(snow_brotli LANGUAGES C)
# Brotli 1.1.0's upstream CMake always builds a CLI. Build only its three static libraries.
foreach(PART common dec enc)
  file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${SNOW_SOURCE}/c/${PART}/*.c")
  add_library(brotli${PART} STATIC ${SOURCES})
  target_include_directories(brotli${PART} PUBLIC "${SNOW_SOURCE}/c/include")
  target_compile_definitions(brotli${PART} PRIVATE BROTLI_HAVE_LOG2=1)
  install(TARGETS brotli${PART} ARCHIVE DESTINATION lib)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/libbrotli${PART}.pc" "prefix=${CMAKE_INSTALL_PREFIX}\nlibdir=\${prefix}/lib\nincludedir=\${prefix}/include\nName: libbrotli${PART}\nDescription: Brotli ${PART}\nVersion: 1.1.0\nLibs: -L\${libdir} -lbrotli${PART}\nCflags: -I\${includedir}\n")
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libbrotli${PART}.pc" DESTINATION lib/pkgconfig)
endforeach()
target_link_libraries(brotlienc PUBLIC brotlicommon)
target_link_libraries(brotlidec PUBLIC brotlicommon)
install(DIRECTORY "${SNOW_SOURCE}/c/include/brotli" DESTINATION include)
